var (

)

type countSorter struct {
	model1.RowEventSorter
	n *int
}

func (c countSorter) Less(i, j int) bool {
	*c.n++
	return c.RowEventSorter.Less(i, j)
}

func sortSort(s sort.Interface) { sort.Sort(s) }

func loadModel(b *testing.B) *model1.TableData {
	once.Do(func() {
		raw, err := os.ReadFile(podsJSON)
		if err != nil {
			b.Fatal(err)
		}
		var list struct{ Items []map[string]any }
		if err := json.Unmarshal(raw, &list); err != nil {
			b.Fatal(err)
		}
		oo := make([]runtime.Object, 0, len(list.Items)*copies)
		for c := range copies {
			for _, it := range list.Items {
				u := &unstructured.Unstructured{Object: runtime.DeepCopyJSON(it)}
				u.SetNamespace(fmt.Sprintf("team-%02d", c+1))
				oo = append(oo, &render.PodWithMetrics{Raw: u})
			}
		}
		r := render.NewPod()
		td := model1.NewTableData(client.PodGVR)
		td.SetHeader(client.NamespaceAll, r.Header(client.NamespaceAll))
		td.Reset(client.NamespaceAll)
		if err := td.Render(context.Background(), r, oo); err != nil {
			b.Fatal(err)
		}
		// second render so rows carry Update/Unchanged kinds like a steady-state tick
		if err := td.Render(context.Background(), r, oo); err != nil {
			b.Fatal(err)
		}
		fixture = td
	})
	b.Logf("rows=%d cols=%d", fixture.RowCount(), fixture.HeaderCount())

	return fixture
}

func newBenchTable(b *testing.B) *Table {
	ctx := context.WithValue(context.Background(), internal.KeyStyles, config.NewStyles())
	ctx = context.WithValue(ctx, internal.KeyViewConfig, config.NewCustomView())
	t := NewTable(client.PodGVR)
	t.Init(ctx)
	t.SetColorerFn(render.NewPod().ColorerFunc())
	t.SetSortCol("NAMESPACE", true)
	t.model.SetNamespace(client.NamespaceAll)
	loadModel(b)

	return t
}

func newScreen(b *testing.B) tcell.SimulationScreen {

	if err := s.Init(); err != nil {
		b.Fatal(err)
	}
	s.SetSize(scrW, scrH)

	return s
}

func BenchmarkLessCalls(b *testing.B) {
	data := loadModel(b).Clone()
	for _, col := range []string{"NAMESPACE", "AGE"} {
		h, idx := data.HeadCol(col, true)
		n := 0
		s := model1.RowEventSorter{Events: data.GetRowEvents(), Index: idx, IsDuration: h.Time, Asc: true}
		cs := countSorter{s, &n}
		sortSort(cs)
		b.Logf("col=%s less_calls_unsorted->sorted=%d", col, n)
		n = 0
		sortSort(cs)
		b.Logf("col=%s less_calls_already_sorted=%d", col, n)
	}
	for b.Loop() {
	}
}

// Model order as Peek returns it each tick (informer list order, unsorted).
func BenchmarkSortFromModelOrder(b *testing.B) {
	base := loadModel(b)
	for _, col := range []string{"NAMESPACE", "AGE"} {
		b.Run(col, func(b *testing.B) {
			b.ReportAllocs()
			for b.Loop() {
				b.StopTimer()
				d := base.Clone()
				b.StartTimer()
				d.Sort(model1.SortColumn{Name: col, ASC: true})
			}
		})
	}
}

func BenchmarkSortShuffled(b *testing.B) {
	base := loadModel(b)
	ee := make([]model1.RowEvent, 0, base.RowCount())
	base.RowsRange(func(_ int, re model1.RowEvent) bool { ee = append(ee, re); return true })
	rand.New(rand.NewSource(1)).Shuffle(len(ee), func(i, j int) { ee[i], ee[j] = ee[j], ee[i] })
	for _, col := range []string{"NAMESPACE", "NAME", "AGE"} {
		b.Run(col, func(b *testing.B) {
			for b.Loop() {
				b.StopTimer()
				cp := append([]model1.RowEvent(nil), ee...)
				d := model1.NewTableDataFull(client.PodGVR, client.NamespaceAll, base.Header(), model1.NewRowEventsWithEvts(cp...))
				b.StartTimer()
				d.Sort(model1.SortColumn{Name: col, ASC: true})
			}
		})
	}
}
